* 200. 岛屿数量

#+BEGIN_SRC scheme :session
(define (num-islands grid)
  (let* ([grid (list->vector (map list->vector grid))]
         [m    (vector-length grid)]
         [n    (vector-length (vector-ref grid 0))])
    (define (dfs i j)
      (cond [(< i 0) 0]
            [(>= i m) 0]
            [(< j 0) 0]
            [(>= j n) 0]
            [(not (char=? #\1 (vector-ref (vector-ref grid i) j))) 0]
            [else (vector-set! (vector-ref grid i) j #\2)
                  (dfs i (- j 1))
                  (dfs i (+ j 1))
                  (dfs (- i 1) j)
                  (dfs (+ i 1) j)]))
    (do ((i 0 (+ i 1))
         (ans 0
              (do ((j 0 (+ j 1))
                   (ans ans
                        (if (char=? (vector-ref (vector-ref grid i) j) #\1)
                            (begin
                              (dfs i j)
                              (+ ans 1))
                            ans)))
                  ((= j n) ans))))
        ((= i m) ans))))
#+END_SRC

* 695. 岛屿的最大面积

(define (max-area-of-island grid)
  (let* ([grid (list->vector (map list->vector grid))]
         [m    (vector-length grid)]
         [n    (vector-length (vector-ref grid 0))])
    (define (dfs i j)
      (cond [(< i 0) 0]
            [(>= i m) 0]
            [(< j 0) 0]
            [(>= j n) 0]
            [(not (= 1 (vector-ref (vector-ref grid i) j))) 0]
            [else (vector-set! (vector-ref grid i) j 0)
                  (+ 1
                     (dfs i (- j 1))
                     (dfs i (+ j 1))
                     (dfs (- i 1) j)
                     (dfs (+ i 1) j))]))
    (for*/fold ([area 0])
               ([i (in-range m)]
                [j (in-range n)])
      (max area (dfs i j)))))

* 3619. 总价值可以被 K 整除的岛屿数目

(define (count-islands grid k)
  (let* ([grid (list->vector (map list->vector grid))]
         [m    (vector-length grid)]
         [n    (vector-length (vector-ref grid 0))])
    (define (dfs i j)
      (cond [(< i 0) 0]
            [(>= i m) 0]
            [(< j 0) 0]
            [(>= j n) 0]
            [(= 0 (vector-ref (vector-ref grid i) j)) 0]
            [else
             (let ([value (vector-ref (vector-ref grid i) j)])
               (vector-set! (vector-ref grid i) j 0)
               (+ value
                  (dfs i (- j 1))
                  (dfs i (+ j 1))
                  (dfs (- i 1) j)
                  (dfs (+ i 1) j)))]))
    (for*/fold ([count 0])
               ([i (in-range m)]
                [j (in-range n)])
      (if (and (not (= 0 (vector-ref (vector-ref grid i) j)))
               (= 0 (modulo (dfs i j) k)))
          (+ count 1)
          count))))

(define (island-perimeter grid)
  (let* ([grid (list->vector (map list->vector grid))]
         [m (vector-length grid)]
         [n (vector-length (vector-ref grid 0))])
    (define (count i j)
      (cond [(< i 0) 1]
            [(>= i m) 1]
            [(< j 0) 1]
            [(>= j n) 1]
            [(= 0 (vector-ref (vector-ref grid i) j)) 1]
            [(= 2 (vector-ref (vector-ref grid i) j)) 0]
            [else (begin
                    (vector-set! (vector-ref grid i) j 2)
                    (+ (count i (+ j 1))
                       (count i (- j 1))
                       (count (+ i 1) j)
                       (count (- i 1) j)))]))
    (for*/fold ([ans 0])
               ([i (in-range m)]
                [j (in-range n)]
                #:when (= 1 (vector-ref (vector-ref grid i) j)))
      (+ ans (count i j)))))
#+END_SRC
